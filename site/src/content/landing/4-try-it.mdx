```js
import { withAUDR } from '@audr/openrouter'

const router = withAUDR(openrouter, {
  customer_id: req.customerId,
  environment: 'production',
  initiator:   'end_user',
  sink:        'file://./audr.jsonl'
})
// nothing else changes; every call now emits a record
```
